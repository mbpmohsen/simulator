// ---------------------------------------------------------------------
// 2. Helper to call the tRPC-style admin endpoints (plain fetch)
// ---------------------------------------------------------------------
async function api<T>(path: string, init?: RequestInit): Promise<T> {
	const res = await fetch(path, {
		...init,
		headers: { "Content-Type": "application/json", ...init?.headers },
	});
	if (!res.ok) {
		const err = await res.json();
		throw new Error(err.error ?? `HTTP ${res.status}`);
	}
	return res.json();
}

											</div>
											<div>
												<Label>شروع دور</Label>
												<Input
													type="number"
													{...regEvent("events.0.start_turn", {
														valueAsNumber: true,
													})}
												/>
											</div>
											<div>
												<Label>مدت (دور)</Label>
												<Input
													type="number"
													{...regEvent("events.0.duration", {
														valueAsNumber: true,
													})}
												/>
											</div>
										</div>

										<div className="grid grid-cols-2 gap-4">
											<div>
												<Label>نوع تغییر</Label>
												<Controller
													control={control}
													name="events.0.modifier_type"
													render={({ field }) => (
														<Select
															onValueChange={field.onChange}
															value={field.value}
														>
															<SelectTrigger>
																<SelectValue />
															</SelectTrigger>
															<SelectContent>
																<SelectItem value="increase">افزایش</SelectItem>
																<SelectItem value="decrease">کاهش</SelectItem>
																<SelectItem value="multiply">ضرب</SelectItem>
															</SelectContent>
														</Select>
													)}
												/>
											</div>
											<div>
												<Label>طرف‌های تحت تأثیر (اختیاری، JSON array)</Label>
												<Input
													placeholder='["قرمز"]'
													{...regEvent("events.0.affected_sides")}
												/>
											</div>
										</div>

										<div className="flex gap-2 justify-end">
											<Button
												type="button"
												variant="secondary"
												onClick={handleEventSubmit(onReplaceEvents)}
												disabled={isSubmitting}
											>
												جایگزینی رویدادها
											</Button>
											<Button
												type="button"
												onClick={handleEventSubmit(onAddEvents)}
												disabled={isSubmitting}
											>
												افزودن رویداد
											</Button>
										</div>
									</form>
								</CardContent>
							</Card>
						</div>
					</TabsContent>

					{/* ======================  شروع بازی  ====================== */}
					<TabsContent value="start">
						<Card>
							<CardHeader>
								<CardTitle>شروع بازی</CardTitle>
								<CardDescription>
									بعد از اطمینان از تنظیمات کلی و رویدادها، بازی را شروع کنید.
								</CardDescription>
							</CardHeader>
							<CardContent className="flex justify-center">
								<Button size="lg" onClick={startGame} disabled={isSubmitting}>
									{isSubmitting ? "در حال شروع..." : "شروع بازی"}
								</Button>
							</CardContent>
						</Card>
					</TabsContent>
				</Tabs>
			</div>
		</div>
	);
